STM32中使用I2C驱动EEPROM读写

背景&需求

产品有一块eeprom存储器件，存放一些基础内容，SN号，location等信息，但是考虑到存储器件不浪费，并且能够利用这些存储内容去存放一些整板维修的记录，在letter-shell的基础上添加命令，进行eeprom的读写。1024字节我写入后面512字节，8条维修记录，一条在64字节内。

eeprom型号：AT24C08C

stm32型号：STM32F1003

I2C_StatusTypeDef I2C_512WriteBytes(uint8_t dev_addr, uint16_t reg_addr, uint8_t *data, uint16_t len)
{
    I2C_StatusTypeDef status;
    uint16_t i;
		
		uint8_t highaddr,lowaddr;
		highaddr = (reg_addr >> 8) & 0xFF;
		lowaddr = reg_addr & 0xFF;
	
		dev_addr |= highaddr;
	
    I2C_Start();
    I2C_SendByte(dev_addr << 1);    // 写地址
    status = I2C_WaitAck();
    if(status != I2C_OK) goto error;
    
    I2C_SendByte(reg_addr);         // 发送寄存器地址
    status = I2C_WaitAck();
    if(status != I2C_OK) goto error;
    
    for(i = 0; i < len; i++)
    {
        I2C_SendByte(data[i]);      // 发送数据
        status = I2C_WaitAck();
        if(status != I2C_OK) goto error;
    }
    
error:
    I2C_Stop();
    return status;
}

const void simple_delay_ms(uint32_t ms) {
    volatile uint32_t i, j;
    for (i = 0; i < ms; i++) {
        for (j = 0; j < 2000; j++) {  // 2000次循环约 0.5~1ms（主频不同请调整）
            __NOP();  // 空操作，防止被优化
        }
    }
}

I2C_StatusTypeDef AT24C08C_Write_CrossPage(uint8_t dev_addr, uint16_t mem_addr, uint8_t *data, uint16_t len) {
    uint16_t offset = 0;
    I2C_StatusTypeDef status = I2C_OK;

    while (len > 0) {
        // 计算当前页内的剩余空间
        uint16_t page_remain = AT24C08C_PAGE_SIZE - (mem_addr % AT24C08C_PAGE_SIZE);
        
        // 本次可以写入的字节数，取“页剩余”和“待写长度”的小值
        uint16_t write_len = (len > page_remain) ? page_remain : len;

        // 调用你的底层I2C页写入函数
        status = I2C_512WriteBytes(dev_addr, mem_addr, &data[offset], write_len);
        if (status != I2C_OK) {
            return status; // 写入失败，立即返回错误码
        }

        // 等待当前页的内部写周期完成 (AT24C08C写周期最大5ms，此处延时留有余量)
        simple_delay_ms(6);
        
        // 更新地址、偏移和剩余长度
        mem_addr += write_len;
        offset += write_len;
        len -= write_len;
    }
    return I2C_OK;
}
